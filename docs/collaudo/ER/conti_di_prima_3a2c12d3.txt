  test('il perche\' proprio lui, prima', () {
    const segno = Zodiac.gemini;
    final nascita = DateTime(1990, 6, 15);
    // Lo stesso cielo di SchermataDelGemello.cieloPer di dopo, che sul
    // commit di partenza non esiste ancora: stessi argomenti.
    final cielo = CieloDiSinastria.perNascita(
      momentoUtc: DateTime.utc(nascita.year, nascita.month, nascita.day, 12),
      oraNota: false,
      latitudine: null,
      longitudineDelLuogo: null,
      segnoDichiarato: segno,
    );
    final titolo = RegExp(r'\b[A-Z][a-zà-ù]+ (?:Congiunzione|Sestile|'
        r'Quadratura|Trigono|Opposizione) [A-Z][a-zà-ù]+');
    final colpe = <String>[];
    var conAspetti = 0;
    for (final v in VipCatalog.vips) {
      final r = SynastryReport.perCieli(
          tuo: cielo, vip: v, quando: DateTime(2026, 9, 28));
      if (r.aspettiPiuForti.isNotEmpty) conAspetti++;
      final t = PercheProprioLui.tecnica(r, segno, v.sign);
      final m = titolo.firstMatch(t);
      if (m != null) colpe.add('${v.name}: "${m.group(0)}"');
    }
    // ignore: avoid_print
    print('PRIMA 3a2c12d3, PERCHE PROPRIO LUI: VIP ${VipCatalog.vips.length}, '
        'con aspetti $conAspetti, titoli dentro la frase ${colpe.length}; '
        'primi tre: ${colpe.take(3).join('; ')}');
  });

  test('ER.17, prima', () {
    final tuo = CieloDiSinastria.perVip(VipCatalog.vips[9]);

    final dueVolte = <String>[];
    for (final vip in VipCatalog.vips) {
      if (vip.luogoDiOggi != null || vip.eScomparso) continue;
      senzaLuogo++;

      final volte =
          'la distanza non entra nel conto'.allMatches(responso).length;
      if (volte != 1) dueVolte.add('${vip.name}: $volte');

  });
