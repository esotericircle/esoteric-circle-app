// IL PERCORSO DELLE SCHERMATE PRINCIPALI SUL SIMULATORE DI IPHONE.
// Ordine GA voce 12, 20 ottobre 2026.
//
// Questo file NON e' una prova della suite: ha l'estensione .txt apposta,
// perche' integration_test non sta nel pubspec del repository (tolto il 6
// ottobre 2026, rompeva le build di rilascio) e un file .dart che lo importa
// romperebbe flutter analyze e flutter test del cancello. Il lavoro
// .github/workflows/iphone-simulatore.yml lo copia in integration_test/,
// aggiunge integration_test e flutter_driver solo sulla copia del giro e lo
// fa girare con flutter drive sul simulatore.
//
// Cosa fa: segna l'onboarding come gia' fatto (solo nelle preferenze locali
// del simulatore, nessun dato di produzione), apre l'app vera dal suo main,
// aspetta che l'intro finisca, fotografa il Santuario, poi apre una per una
// le schermate principali con le loro rotte vere, quelle che usa
// art_navigation.dart, e le fotografa. Ogni schermata e' presa in un blocco
// suo: se una cade, le altre si fotografano lo stesso, e il registro dice
// quale e perche'.
//
// Le catture arrivano per due strade, cosi' una basta: al driver sul Mac
// (test_driver/integration_test.dart, che le scrive nella cartella delle
// catture) e nella cartella dei documenti dell'app, che il lavoro copia fuori
// con xcrun simctl get_app_container.
import 'dart:async';
import 'dart:io';
import 'dart:ui';

import 'package:esoteric_circle/core/astro/zodiac.dart';
import 'package:esoteric_circle/features/angels/oracolo/oracolo_degli_angeli_screen.dart';
import 'package:esoteric_circle/features/horoscope/oroscopo_screen.dart';
import 'package:esoteric_circle/features/maestri/caligo/rune/rune_draw_screen.dart';
import 'package:esoteric_circle/features/real_time_cosmo/cielo_reale_screen.dart';
import 'package:esoteric_circle/features/tarot/stesa_tre_carte_screen.dart';
import 'package:esoteric_circle/main.dart' as app;
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:path_provider/path_provider.dart';
import 'package:shared_preferences/shared_preferences.dart';

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();

  Future<void> aspetta(WidgetTester tester, int secondi) async {
    for (var i = 0; i < secondi * 4; i++) {
      await tester.pump(const Duration(milliseconds: 250));
      await Future<void>.delayed(const Duration(milliseconds: 250));
    }
    // Un guasto dell'app durante l'attesa non deve far cadere il percorso:
    // si stampa e si va avanti, la cattura dice il resto.
    final guasto = tester.takeException();
    if (guasto != null) print('PERCORSO GA.12: guasto durante l\'attesa: $guasto');
  }

  Future<void> scatta(WidgetTester tester, String nome) async {
    try {
      final byte = await binding.takeScreenshot(nome);
      final cartella = await getApplicationDocumentsDirectory();
      final file = File('${cartella.path}/catture_percorso/$nome.png');
      file.parent.createSync(recursive: true);
      file.writeAsBytesSync(byte);
      print('PERCORSO GA.12: cattura $nome, ${byte.length} byte');
    } catch (e) {
      print('PERCORSO GA.12: cattura $nome NON riuscita: $e');
    }
  }

  testWidgets('le schermate principali si aprono e si fotografano',
      (tester) async {
    final erroreDiPrima = FlutterError.onError;
    final piattaformaDiPrima = PlatformDispatcher.instance.onError;

    final preferenze = await SharedPreferences.getInstance();
    await preferenze.setBool('onboarding.done', true);

    unawaited(app.main());
    await aspetta(tester, 30);
    await scatta(tester, '00_santuario');

    final navigatore =
        tester.state<NavigatorState>(find.byType(Navigator).first);
    final rotte = <String, Route<void> Function()>{
      '01_oroscopo': () => OroscopoScreen.route(userSign: Zodiac.gemini),
      '02_stesa_dei_tarocchi': () => StesaTreCarteScreen.route(),
      '03_oracolo_degli_angeli': () => OracoloDegliAngeliScreen.route(),
      '04_rune': () => RuneDrawScreen.route(),
      '05_cielo_in_tempo_reale': () =>
          CieloRealeScreen.route(ModoDelCielo.adesso),
    };
    for (final voce in rotte.entries) {
      try {
        unawaited(navigatore.push(voce.value()));
        await aspetta(tester, 10);
        await scatta(tester, voce.key);
        navigatore.pop();
        await aspetta(tester, 3);
      } catch (e, pila) {
        print('PERCORSO GA.12: ${voce.key} NON aperta: $e\n$pila');
      }
    }

    // L'app sostituisce i gestori degli errori con Crashlytics: si rimettono
    // quelli della prova, altrimenti la prova cade alla fine per questo.
    FlutterError.onError = erroreDiPrima;
    PlatformDispatcher.instance.onError = piattaformaDiPrima;
  });
}
